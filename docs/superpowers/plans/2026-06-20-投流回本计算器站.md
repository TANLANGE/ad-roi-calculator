Create `d:\Code\网站2\src\lib\calculators.ts`:

```ts
import type { PaybackInput, ProfitInput, RoiInput } from '@/types/calculator';

const round = (value: number) => Number(value.toFixed(1));

export function calculateRoi(input: RoiInput) {
  const revenue = input.orders * input.averageOrderValue;
  const grossProfit = revenue * (input.grossMarginRate / 100);
  const netProfit = grossProfit - input.adSpend;
  const roiRate = input.adSpend === 0 ? 0 : (netProfit / input.adSpend) * 100;
  const customerAcquisitionCost = input.orders === 0 ? 0 : input.adSpend / input.orders;

  return {
    revenue,
    grossProfit,
    netProfit,
    roiRate: round(roiRate),
    customerAcquisitionCost: round(customerAcquisitionCost),
  };
}

export function calculatePayback(input: PaybackInput) {
  const revenue = input.monthlyOrders * input.averageOrderValue;
  const grossProfit = revenue * (input.grossMarginRate / 100);
  const monthlyNetProfit = grossProfit - input.monthlyAdSpend;
  const paybackMonths =
    monthlyNetProfit <= 0 ? Infinity : Math.ceil(input.initialInvestment / monthlyNetProfit);

  const cumulativeProfitPoints = Array.from({ length: 12 }, (_, index) => ({
    month: index + 1,
    profit: monthlyNetProfit * (index + 1) - input.initialInvestment,
  }));

  return {
    monthlyNetProfit,
    paybackMonths,
    cumulativeProfitPoints,
  };
}

export function calculateProfit(input: ProfitInput) {
  const revenue = input.sellingPrice * input.quantity;
  const totalCost = input.unitCost * input.quantity;
  const grossProfit = revenue - totalCost;
  const netProfit = grossProfit - input.adSpend;
  const grossMarginRate = revenue === 0 ? 0 : (grossProfit / revenue) * 100;
  const unitProfit = input.quantity === 0 ? 0 : netProfit / input.quantity;

  return {
    revenue,
    totalCost,
    grossProfit,
    netProfit,
    grossMarginRate: round(grossMarginRate),
    unitProfit: round(unitProfit),
  };
}
```
